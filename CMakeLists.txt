cmake_minimum_required(VERSION 3.25)
project(AirportControlTower LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Crow CONFIG REQUIRED)
add_executable(airport-control-tower backend/src/main.cpp)
target_include_directories(airport-control-tower PRIVATE backend/include)
target_link_libraries(airport-control-tower PRIVATE Crow::Crow)
if(MSVC)
  target_compile_options(airport-control-tower PRIVATE /W4 /permissive-)
endif()
