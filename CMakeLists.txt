cmake_minimum_required(VERSION 3.25)
project(AirportControlTower LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Crow CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(airport-control-tower backend/src/main.cpp)
target_include_directories(airport-control-tower PRIVATE backend/include)
target_link_libraries(airport-control-tower PRIVATE Crow::Crow Threads::Threads)
if(MSVC)
  target_compile_options(airport-control-tower PRIVATE /W4 /permissive-)
  target_compile_definitions(airport-control-tower PRIVATE _WIN32_WINNT=0x0A00)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(airport-simulation-tests backend/tests/simulation_tests.cpp)
  target_include_directories(airport-simulation-tests PRIVATE backend/include)
  target_link_libraries(airport-simulation-tests PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(airport-simulation-tests PRIVATE /W4 /permissive-)
  endif()
  add_test(NAME airport-simulation-unit-and-integration COMMAND airport-simulation-tests)
endif()
